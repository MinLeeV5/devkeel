交互 JS

// 折叠/展开 + localStorage 持久化
function toggleSection(head) {
  const section = head.parentElement;
  section.classList.toggle('collapsed');
  localStorage.setItem(section.id, section.classList.contains('collapsed'));
}

// Artifact 标签页 + 按需 marked.js 渲染
function switchArtifactTab(name) {
  document.querySelectorAll('.artifact-tab').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.artifact-panel').forEach(p => p.classList.remove('active'));
  document.querySelector(`[data-tab="${name}"]`).classList.add('active');
  const panel = document.querySelector(`[data-panel="${name}"]`);
  panel.classList.add('active');
  if (!panel.dataset.rendered) {
    const md = document.querySelector(`script[data-artifact="${name}"]`).textContent;
    panel.innerHTML = typeof marked !== 'undefined' ? marked.parse(md) : `
${md}
`;
    panel.dataset.rendered = 'true';
  }
}
